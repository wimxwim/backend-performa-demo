// 01: no logger.ts — console.log langsung
// 02: pino JSON SOP
import pino from "pino";
export const logger = pino({
  level: process.env.LOG_LEVEL || "info",
  formatters: { level: (l) => ({ level: l }) },
  timestamp: pino.stdTimeFunctions.isoTime,
});
// usage: logger.info({ requestId, orderId }, "order created")
compose.yaml — scale 03 vs centralized 04 
# 03 Pasar 6.081 — scale
services:
  order-service:
    deploy: { replicas: 3 }
  nginx: { image: nginx, ports: ["3000:80"] }

# 04 SAKTI 5M — centralized
services:
  alloy: { image: grafana/alloy, volumes: ["./config.alloy:/etc/alloy/config.alloy"] }
  loki: { image: grafana/loki }
  grafana: { image: grafana/grafana, ports: ["3001:3000"] }
config.alloy — Alloy → Loki 
loki.source.file "warung" {
  targets = [{ __path__ = "/var/log/warung/*.log", job = "warung-service" }]
  forward_to = [loki.write.grafana.receiver]
}
loki.write "grafana" {
  endpoint { url = "http://loki:3100/loki/api/v1/push" }
}
